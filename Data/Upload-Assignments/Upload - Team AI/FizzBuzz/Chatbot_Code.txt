Ein einfaches Tic Tac Toe-Spiel ist eine großartige Möglichkeit, die Grundlagen von Python zu lernen! Das Grundgerüst könnte so aussehen:

def print_board(board):
    for row in board:
        print("|".join(row))
        print("-" * 5)

def check_winner(board):
    # Überprüfen Sie die Zeilen, Spalten und Diagonalen auf einen Gewinner
    for row in board:
        if row[0] == row[1] == row[2] != " ":
            return row[0]
    for col in range(3):
        if board[0][col] == board[1][col] == board[2][col] != " ":
            return board[0][col]
    if board[0][0] == board[1][1] == board[2][2] != " ":
        return board[0][0]
    if board[0][2] == board[1][1] == board[2][0] != " ":
        return board[0][2]
    return None

def tic_tac_toe():
    board = [[" " for _ in range(3)] for _ in range(3)]
    current_player = "X"

    for _ in range(9):
        print_board(board)
        row = int(input(f"Spieler {current_player}, wählen Sie eine Zeile (0-2): "))
        col = int(input(f"Spieler {current_player}, wählen Sie eine Spalte (0-2): "))

        if board[row][col] == " ":
            board[row][col] = current_player
            if check_winner(board):
                print_board(board)
                print(f"Spieler {current_player} hat gewonnen!")
                return
            current_player = "O" if current_player == "X" else "X"
        else:
            print("Ungültiger Zug! Bitte versuchen Sie es erneut.")

    print_board(board)
    print("Das Spiel endet Unentschieden!")

tic_tac_toe()
